```ts
import type { EventOut, VenueOut } from './types'

export interface EventQuery {
  date_from?: string
  date_to?: string
  genre?: string
  min_confidence?: number
  bbox?: string
  include_past?: boolean
}

export interface VenueQuery {
  type?: string
  min_confidence?: number
  bbox?: string
}

export function buildQuery(params: Record<string, string | number | boolean | undefined>): string {
  const parts: string[] = []
  for (const [key, value] of Object.entries(params)) {
    if (value === undefined || value === '' || value === null) continue
    parts.push(`${key}=${encodeURIComponent(String(value))}`)
  }
  return parts.length ? `?${parts.join('&')}` : ''
}

async function getJson<T>(url: string): Promise<T> {
  const resp = await fetch(url)
  if (!resp.ok) throw new Error(`API ${resp.status} em ${url}`)
  return resp.json() as Promise<T>
}

export function fetchEvents(q: EventQuery): Promise<EventOut[]> {
  return getJson<EventOut[]>(`/api/events${buildQuery(q)}`)
}

export function fetchVenues(q: VenueQuery = {}): Promise<VenueOut[]> {
  return getJson<VenueOut[]>(`/api/venues${buildQuery(q)}`)
}
```

```ts
import type { EventOut, VenueOut } from '../api/types'

export interface EventWithGeo extends EventOut {
  venueName: string | null
  lat: number | null
  lng: number | null
  mappable: boolean
}

export function joinEventsWithVenues(events: EventOut[], venues: VenueOut[]): EventWithGeo[] {
  const byId = new Map(venues.map((v) => [v.id, v]))
  return events.map((e) => {
    const v = e.venue_id != null ? byId.get(e.venue_id) : undefined
    const lat = v?.lat ?? null
    const lng = v?.lng ?? null
    return {
      ...e,
      venueName: v?.name ?? null,
      lat,
      lng,
      mappable: lat != null && lng != null,
    }
  })
}
```

```ts
import type { EventQuery } from '../api/client'

export type Preset = 'upcoming' | 'today' | 'weekend' | 'all'

export interface FilterState {
  preset: Preset
  genre: string | null
  minConfidence: number
  includePast: boolean
}

export const initialFilters: FilterState = {
  preset: 'upcoming', genre: null, minConfidence: 0, includePast: false,
}

export type FilterAction =
  | { type: 'setPreset'; preset: Preset }
  | { type: 'setGenre'; genre: string | null }
  | { type: 'setMinConfidence'; value: number }
  | { type: 'setIncludePast'; value: boolean }

export function filterReducer(state: FilterState, action: FilterAction): FilterState {
  switch (action.type) {
    case 'setPreset': return { ...state, preset: action.preset, includePast: action.preset === 'all' }
    case 'setGenre': return { ...state, genre: action.genre }
    case 'setMinConfidence': return { ...state, minConfidence: action.value }
    case 'setIncludePast': return { ...state, includePast: action.value }
  }
}

function isoDate(d: Date): string {
  return d.toISOString().slice(0, 10)
}

export function queryFromFilters(f: FilterState, now: Date = new Date()): EventQuery {
  const q: EventQuery = {}
  if (f.genre) q.genre = f.genre
  if (f.minConfidence > 0) q.min_confidence = f.minConfidence

  if (f.preset === 'today') {
    q.date_from = isoDate(now)
    q.date_to = isoDate(now)
  } else if (f.preset === 'weekend') {
    // proxima sexta ate domingo
    const day = now.getDay() // 0=dom..6=sab
    const toFri = (5 - day + 7) % 7
    const fri = new Date(now); fri.setDate(now.getDate() + toFri)
    const sun = new Date(fri); sun.setDate(fri.getDate() + 2)
    q.date_from = isoDate(fri)
    q.date_to = isoDate(sun)
  } else if (f.preset === 'all') {
    q.include_past = true
  }
  // 'upcoming' => sem datas (a API ja default-filtra futuro)
  return q
}
```

```tsx
import type { EventWithGeo } from '../data/join'

function fmtDate(iso: string | null): string {
  if (!iso) return 'data a confirmar'
  const d = new Date(iso)
  return d.toLocaleString('pt-BR', { weekday: 'short', day: '2-digit', month: '2-digit', hour: '2-digit', minute: '2-digit' })
}

export function EventCard({ event, selected, onSelect }: {
  event: EventWithGeo; selected: boolean; onSelect: (id: number) => void
}) {
  return (
    <div className={`evcard${selected ? ' sel' : ''}`} onClick={() => onSelect(event.id)}>
      <span className="conf">{event.effective_confidence}</span>
      <div className="name">{event.name}</div>
      <div className="meta">
        {fmtDate(event.starts_at)}
        {event.venueName ? ` · ${event.venueName}` : ''}
        {!event.mappable && <span className="nogeo"> · local a confirmar</span>}
      </div>
      <div>
        {event.genre && <span className="tag">{event.genre}</span>}
```

```tsx
import { useEffect, useMemo, useReducer, useState } from 'react'
import './theme.css'
import { fetchEvents, fetchVenues } from './api/client'
import { joinEventsWithVenues, type EventWithGeo } from './data/join'
import { filterReducer, initialFilters, queryFromFilters } from './filters/state'
import { FilterBar, type View } from './filters/FilterBar'
import { EventList } from './list/EventList'
import { MapView } from './map/MapView'

export default function App() {
  const [filters, dispatch] = useReducer(filterReducer, initialFilters)
  const [view, setView] = useState<View>('split')
  const [events, setEvents] = useState<EventWithGeo[]>([])
  const [selectedId, setSelectedId] = useState<number | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    let alive = true
    setLoading(true); setError(null)
    Promise.all([fetchEvents(queryFromFilters(filters)), fetchVenues()])
      .then(([evs, vns]) => { if (alive) setEvents(joinEventsWithVenues(evs, vns)) })
      .catch((e) => { if (alive) setError(String(e)) })
      .finally(() => { if (alive) setLoading(false) })
    return () => { alive = false }
  }, [filters])

  const genres = useMemo(
    () => Array.from(new Set(events.map((e) => e.genre).filter((g): g is string => !!g))).sort(),
    [events],
  )

  return (
    <div style={{ height: '100vh', display: 'flex', flexDirection: 'column' }}>
      <FilterBar filters={filters} genres={genres} dispatch={dispatch} view={view} onView={setView} />
      {error && <div className="empty" style={{ color: '#e07dd0' }}>Não consegui falar com a API — o uvicorn está no ar? ({error})</div>}
      {loading && <div className="empty">Carregando…</div>}
      {!loading && !error && (
        <div style={{ flex: 1, display: 'flex', minHeight: 0 }}>
          <div style={{ width: view === 'split' ? '42%' : '100%', overflowY: 'auto' }}>
            <EventList events={events} selectedId={selectedId} onSelect={setSelectedId} />
          </div>
          {view === 'split' && (
            <div style={{ flex: 1 }}>
              <MapView events={events} selectedId={selectedId} onSelect={setSelectedId} />
            </div>
          )}
        </div>
      )}
    </div>
  )
}
```
