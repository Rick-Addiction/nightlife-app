```python
# src/nightlife/ingestion/web_source.py
"""Fetch de site de venue + descoberta da pagina de agenda (Ciclo E-2a).
Parsers PUROS (agenda_links/looks_empty/visible_text) testados; fetch_page e
shell smoke-calibrado (httpx -> Playwright de fallback)."""
from __future__ import annotations

import re
import unicodedata
from html.parser import HTMLParser
from urllib.parse import urljoin, urlparse

_AGENDA_WORDS = ("agenda", "eventos", "evento", "programacao", "programa", "shows", "line")
_JSONLD_RE = re.compile(r'type=["\']application/ld\+json["\']', re.I)
_TEXT_MIN = 200          # abaixo disso + sem jsonld => provavelmente JS-render vazio


def _norm(s: str) -> str:
    return "".join(c for c in unicodedata.normalize("NFD", s.lower())
                   if unicodedata.category(c) != "Mn")


class _TextExtractor(HTMLParser):
    def __init__(self):
        super().__init__()
        self._skip = 0
        self.parts: list[str] = []
    def handle_starttag(self, tag, attrs):
        if tag in ("script", "style"):
            self._skip += 1
    def handle_endtag(self, tag):
        if tag in ("script", "style") and self._skip:
            self._skip -= 1
    def handle_data(self, data):
        if not self._skip and data.strip():
            self.parts.append(data.strip())


def visible_text(html: str) -> str:
    p = _TextExtractor()
    try:
        p.feed(html)
    except Exception:  # noqa: BLE001 — html quebrado: devolve o que juntou
        pass
    return " ".join(p.parts)


def looks_empty(html: str) -> bool:
    if _JSONLD_RE.search(html):
        return False
    return len(visible_text(html)) < _TEXT_MIN


class _LinkGrabber(HTMLParser):
    def __init__(self):
        super().__init__()
        self.links: list[tuple[str, str]] = []   # (href, texto)
        self._href = None
        self._buf: list[str] = []
    def handle_starttag(self, tag, attrs):
        if tag == "a":
            self._href = dict(attrs).get("href")
            self._buf = []
    def handle_data(self, data):
        if self._href is not None:
            self._buf.append(data)
    def handle_endtag(self, tag):
        if tag == "a" and self._href:
            self.links.append((self._href, " ".join(self._buf)))
            self._href = None


def agenda_links(html: str, *, base_url: str, cap: int = 3) -> list[str]:
    g = _LinkGrabber()
    try:
        g.feed(html)
    except Exception:  # noqa: BLE001
        pass
    host = urlparse(base_url).netloc
    out: list[str] = []
    for href, txt in g.links:
        blob = _norm(f"{href} {txt}")
        if not any(w in blob for w in _AGENDA_WORDS):
            continue
        absu = urljoin(base_url, href)
        if urlparse(absu).netloc != host:      # so mesmo dominio
            continue
        if absu not in out:
            out.append(absu)
        if len(out) >= cap:
            break
    return out


def fetch_page(url: str, *, timeout: float = 15.0) -> str:
    """httpx primeiro; se looks_empty, Playwright renderiza. Shell smoke."""
    import httpx
    import truststore
    truststore.inject_into_ssl()
    headers = {"User-Agent": "RadarSP/1.0 (agenda de eventos; contato via github)"}
    html = ""
    try:
        r = httpx.get(url, headers=headers, timeout=timeout, follow_redirects=True)
        html = r.text
    except Exception:  # noqa: BLE001 — rede/tls: tenta o Playwright
        html = ""
    if html and not looks_empty(html):
        return html
    # fallback renderizado
    try:
        from playwright.sync_api import sync_playwright
        with sync_playwright() as pw:
            b = pw.chromium.launch(headless=True)
            page = b.new_page(user_agent=headers["User-Agent"])
            try:
                page.goto(url, wait_until="networkidle", timeout=int(timeout * 1000))
                page.wait_for_timeout(1500)
                html = page.content()
            finally:
                page.close(); b.close()
    except Exception:  # noqa: BLE001 — sem render: devolve o httpx (mesmo que vazio)
        pass
    return html
```

"""Orquestra a extracao do site de um venue (Ciclo E-2a): fetch -> JSON-LD ou
codex-texto -> acoes ancoradas por venue_id. Cache por hash: site parado pula."""
from __future__ import annotations

import hashlib
from datetime import date, datetime, timezone
from pathlib import Path
from typing import Callable

from sqlalchemy import select
from sqlalchemy.orm import Session

from ..extraction.jsonld import parse_jsonld_events
from ..extraction.post_actions import AddEvent, PostInterpretation, sanitize_actions
from ..models import Event, Venue
from ..persistence.action_executor import execute_actions
from .post_interpreter import Document, VenueContext
from .web_source import agenda_links, looks_empty, visible_text

_SP_TODAY_CAP = 20
_TEXT_CAP = 12000        # trunca texto longo pro prompt (custo/alucinacao)


def jsonld_to_actions(html: str, *, source_url: str | None) -> list[AddEvent]:
    out: list[AddEvent] = []
    for ev in parse_jsonld_events(html, source_url=source_url):
        if not ev.name or not ev.starts_at_raw:
            continue
        out.append(AddEvent(
            name=ev.name, starts_at_iso=ev.starts_at_raw, ticket_url=ev.ticket_url,
            quotes={"name": ev.name}, confidence={"name": 1.0}))
    return out


def _context_for(session: Session, venue: Venue, today: date) -> VenueContext:
    now = datetime.now(timezone.utc)
    known: list[tuple[str, str | None]] = []
    for e in session.scalars(
            select(Event).where(Event.venue_id == venue.id, Event.status != "cancelled")
            .order_by(Event.starts_at.asc().nullslast()).limit(_SP_TODAY_CAP)):
        if e.starts_at is None or e.starts_at >= now:
            known.append((e.name, e.starts_at.date().isoformat() if e.starts_at else None))
    return VenueContext(venue_name=venue.name, handle=venue.instagram,
                        today_iso=today.isoformat(), known_events=known)


def crawl_venue(session: Session, venue: Venue, *, fetcher: Callable[[str], str],
                interpreter, reference: date) -> dict:
    stats = {"venue": venue.name, "add": 0, "jsonld": 0, "codex": 0, "skipped": False}
    if not venue.website:
        return stats
    home = fetcher(venue.website)
    digest = hashlib.sha256((home or "").encode("utf-8", "replace")).hexdigest()
    if venue.web_last_hash == digest:
        stats["skipped"] = True
        return stats

    pages = [(venue.website, home)]
    for link in agenda_links(home, base_url=venue.website):
        pages.append((link, fetcher(link)))

    ctx = _context_for(session, venue, reference)
    for url, html in pages:
        if not html:
            continue
        acts = jsonld_to_actions(html, source_url=url)
        if acts:
            stats["jsonld"] += len(acts)
            interp = PostInterpretation(actions=acts)
        else:
            doc = Document(kind="web_page", image_path=None, caption=None, handle=None,
                           permalink=url, text=visible_text(html)[:_TEXT_CAP], venue_id=venue.id)
            result = interpreter.interpret(doc, ctx)
            if result is None:
                continue
            stats["codex"] += 1
            interp = result
        doc = Document(kind="web_page", image_path=None, caption=None, handle=None,
                       permalink=url, text=None, venue_id=venue.id)
        s = execute_actions(session, doc, sanitize_actions(interp), reference=reference)
        stats["add"] += s["add"]

    venue.web_last_hash = digest
    venue.web_checked_at = datetime.now(timezone.utc)
    session.commit()
    return stats
```

# src/nightlife/crawl_sites.py
"""Varre os venues com SITE PROPRIO e extrai a agenda (Ciclo E-2a). Semanal.
Uso: .venv/Scripts/python -m nightlife.crawl_sites [--limit N]"""
from __future__ import annotations

import argparse
from datetime import date

from sqlalchemy import or_, select
from sqlalchemy.orm import Session

from .db import SessionLocal
from .ingestion.post_interpreter import PostInterpreter
from .ingestion.web_flow import crawl_venue
from .ingestion.web_source import fetch_page
from .models import IngestionRun, Venue

_EXCLUDE = ("%instagram.com%", "%facebook.com%", "%linktr%", "%linktree%")


def own_site_venues(session: Session) -> list[Venue]:
    q = select(Venue).where(Venue.website.is_not(None))
    for pat in _EXCLUDE:
        q = q.where(~Venue.website.ilike(pat))
    return list(session.scalars(q.order_by(Venue.id)))


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--limit", type=int, default=None)
    args = ap.parse_args()
    today = date.today()
    with SessionLocal() as session:
        run = IngestionRun(source="sites", status="running")
        session.add(run); session.flush()
        venues = own_site_venues(session)
        if args.limit:
            venues = venues[: args.limit]
        interpreter = PostInterpreter()
        total_add = found = 0
        for v in venues:
            found += 1
            try:
                stats = crawl_venue(session, v, fetcher=fetch_page,
                                    interpreter=interpreter, reference=today)
                total_add += stats.get("add", 0)
                mark = "PULOU" if stats.get("skipped") else f"+{stats.get('add',0)}"
                print(f"  {v.name[:34]:36} {mark} (jsonld={stats.get('jsonld',0)} codex={stats.get('codex',0)})")
            except Exception as exc:  # noqa: BLE001 — falha isolada por venue
                print(f"  {v.name[:34]:36} ERRO: {str(exc)[:60]}")
        run.status = "finished"; run.records_found = found; run.records_created = total_add
        from datetime import datetime, timezone
        run.finished_at = datetime.now(timezone.utc)
        session.commit()
        print(f"sites: {found} venues, +{total_add} eventos")


if __name__ == "__main__":
    main()
```
